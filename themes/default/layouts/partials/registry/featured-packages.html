{{- /* Returns the featured packages in display order. A package needs `featured: true` in its YAML to appear; this list only orders them, since ranging over site.Data sorts by file name. Four per row on wide screens. */ -}} {{- $order := slice "aws" "azure-native" "gcp" "kubernetes" "cloudflare" "terraform-provider" "pulumiservice" "delivery" -}} {{- $packages := site.Data.registry.packages -}} {{- $featured := slice -}} {{- range $order -}} {{- with index $packages . -}} {{- if .featured -}}{{- $featured = $featured | append . -}}{{- end -}} {{- end -}} {{- end -}} {{- return $featured -}}
